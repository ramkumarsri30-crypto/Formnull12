     3→/**
     4→ * FormNull — Form Builder (Phase 2)
     5→ * =====================================================================
     6→ * The manual form builder. THE SOURCE OF TRUTH for form structure.
     7→ *
     8→ * Real Supabase persistence:
     9→ *   - Form load:      SELECT forms + form_fields (sorted by sort_order)
    10→ *   - Form save:      UPDATE forms (name, description)
    11→ *   - Form delete:    DELETE forms (cascades fields + submissions via FK)
    12→ *   - Add field:      INSERT form_fields (immediate, sort_order = max+1)
    13→ *   - Edit field:     UPDATE form_fields (explicit Save per field)
    14→ *   - Delete field:   DELETE form_fields (cascades submission_values via FK)
    15→ *   - Reorder:        UPDATE sort_order for CHANGED rows only, once per
    16→ *                     drop / button press (no writes during dragging)
    17→ *
    18→ * No fake success: every mutation awaits the real Supabase response and
    19→ * only updates UI state after the database confirms. Failures surface
    20→ * real errors and preserve unsaved local work.
    21→ *
    22→ * RLS: all queries run as the authenticated user — readers must be
    23→ * workspace members, writers editors+. An unauthorized form id simply
    24→ * returns no row ("Form not found").
    25→ */

    80→/* field_key generation — unique per form                              */

    82→
    83→function slugify(s: string): string {
    84→  return (
    85→    s
    86→      .toLowerCase()
    87→      .trim()
    88→      .replace(/[^a-z0-9]+/g, "_")
    89→      .replace(/^_+|_+$/g, "")
    90→      .slice(0, 50) || "field"
    91→  );
    92→}
    93→
    94→function uniqueKey(label: string, existing: string[]): string {
    95→  const base = slugify(label);
    96→  if (!existing.includes(base)) return base;
    97→  let i = 2;
    98→  while (existing.includes(`${base}_${i}`)) i += 1;
    99→  return `${base}_${i}`;
   100→}

   106→export function FormDetail({ formId }: { formId: string }) {
   107→  const router = useRouter();
   108→  const { user } = useAuth();
   109→  const [form, setForm] = useState<FormRow | null>(null);
   110→  const [fields, setFields] = useState<FormField[]>([]);
   111→  const [loading, setLoading] = useState(true);
   112→  const [loadError, setLoadError] = useState<string | null>(null);
   113→
   114→  // Form-level editing
   115→  const [name, setName] = useState("");
   116→  const [description, setDescription] = useState("");
   117→  const [savingForm, setSavingForm] = useState(false);
   118→  const [deleting, setDeleting] = useState(false);
   119→
   120→  // Field-level editing
   121→  const [editingFieldId, setEditingFieldId] = useState<string | null>(null);
   122→  const [savingFieldId, setSavingFieldId] = useState<string | null>(null);
   123→  const [addingField, setAddingField] = useState<string | null>(null);
   124→  const [deletingFieldId, setDeletingFieldId] = useState<string | null>(null);
   125→  const [reorderError, setReorderError] = useState<string | null>(null);
   126→
   127→  const fieldIds = useMemo(() => fields.map((f) => f.id), [fields]);
   128→
   129→  const load = useCallback(async () => {
   130→    setLoading(true);
   131→    setLoadError(null);
   132→    try {
   133→      const [formRes, fieldsRes] = await Promise.all([
   134→        supabaseBrowser.from("forms").select("*").eq("id", formId).maybeSingle(),
   135→        supabaseBrowser
   136→          .from("form_fields")
   137→          .select("*")
   138→          .eq("form_id", formId)
   139→          .order("sort_order", { ascending: true }),
   140→      ]);
   141→
   142→      if (formRes.error) throw formRes.error;
   143→      if (!formRes.data) {
   144→        setForm(null);
   145→        setLoadError("Form not found, or you don't have access to it.");
   146→        setLoading(false);
   147→        return;
   148→      }
   149→      setForm(formRes.data);
   150→      setName(formRes.data.name);
   151→      setDescription(formRes.data.description ?? "");
   152→
   153→      if (fieldsRes.error) throw fieldsRes.error;
   154→      setFields((fieldsRes.data as FormField[]) ?? []);
   155→    } catch (e) {
   156→      const msg = e instanceof Error ? e.message : "Failed to load form.";
   157→      setLoadError(msg);
   158→    } finally {
   159→      setLoading(false);
   160→    }
   161→  }, [formId]);
   162→
   163→  useEffect(() => {
   164→    load();
   165→  }, [load]);

   167→  /* ---------------- Form-level mutations ---------------- */
   168→
   169→  async function saveForm() {
   170→    if (!form) return;
   171→    if (!name.trim()) {
   172→      toast.error("Form name cannot be empty.");
   173→      return;
   174→    }
   175→    // updated_by must be the AUTHENTICATED current user — never a
   176→    // fabricated or inherited id (previously this wrote created_by,
   177→    // which misattributes edits when a workspace admin saves the form).
   178→    if (!user) {
   179→      toast.error("Your session has expired. Please sign in again.");
   180→      return;
   181→    }
   182→    setSavingForm(true);
   183→    const { error } = await supabaseBrowser
   184→      .from("forms")
   185→      .update({
   186→        name: name.trim(),
   187→        description: description.trim() || null,
   188→        updated_by: user.id,
   189→      })
   190→      .eq("id", form.id);
   191→    setSavingForm(false);
   192→    if (error) {
   193→      toast.error("Could not save form.", { description: error.message });
   194→      return;
   195→    }
   196→    toast.success("Form updated.");
   197→    setForm({
   198→      ...form,
   199→      name: name.trim(),
   200→      description: description.trim() || null,
   201→      updated_by: user.id,
   202→    });
   203→  }
   204→
   205→  async function deleteForm() {
   206→    if (!form) return;
   207→    if (!confirm(`Delete "${form.name}"? All fields and submissions will be permanently removed. This cannot be undone.`)) {
   208→      return;
   209→    }
   210→    setDeleting(true);
   211→    const { error } = await supabaseBrowser.from("forms").delete().eq("id", form.id);
   212→    setDeleting(false);
   213→    if (error) {
   214→      toast.error("Could not delete form.", { description: error.message });
   215→      return;
   216→    }
   217→    toast.success("Form deleted.");
   218→    router.push("/dashboard/forms/");
   219→    router.refresh();
   220→  }
   221→
   222→  /* ---------------- Field mutations ---------------- */
   223→
   224→  async function addField(type: FieldType) {
   225→    if (!form) return;
   226→    const meta = { type, label: fieldLabel(type) };
   227→    setAddingField(type);
   228→    try {
   229→      const existingKeys = fields.map((f) => f.field_key);
   230→      const key = uniqueKey(meta.label, existingKeys);
   231→      const nextOrder = fields.length > 0 ? Math.max(...fields.map((f) => f.sort_order)) + 1 : 0;
   232→
   233→      // Type-appropriate default config from the CENTRALIZED registry
   234→      // (selects start with valid options, ratings with max=5).
   235→      const defaultConfig = defaultConfigForType(type);
   236→
   237→      const { data, error } = await supabaseBrowser
   238→        .from("form_fields")
   239→        .insert({
   240→          form_id: form.id,
   241→          field_key: key,
   242→          field_type: type,
   243→          label: meta.label,
   244→          is_required: false,
   245→          sort_order: nextOrder,
   246→          width: 12,
   247→          config: defaultConfig,
   248→        })
   249→        .select()
   250→        .single();
   251→
   252→      if (error) throw error;
   253→      if (!data) throw new Error("Field creation returned no row.");
   254→
   255→      // Real row confirmed by the database — now update the UI.
   256→      setFields((prev) => [...prev, data as FormField]);
   257→      setEditingFieldId(data.id);
   258→      toast.success(`${meta.label} field added.`);
   259→    } catch (e) {
   260→      toast.error("Could not add field.", {
   261→        description: e instanceof Error ? e.message : "Please try again.",
   262→      });
   263→    } finally {
   264→      setAddingField(null);
   265→    }
   266→  }
   267→
   268→  async function saveField(
   269→    fieldId: string,
   270→    draft: {
   271→      label: string;
   272→      description: string | null;
   273→      placeholder: string | null;
   274→      help_text: string | null;
   275→      is_required: boolean;
   276→      width: number;
   277→      config: Record<string, unknown>;
   278→    },
   279→  ): Promise<boolean> {
   280→    setSavingFieldId(fieldId);
   281→    const { error } = await supabaseBrowser
   282→      .from("form_fields")
   283→      .update({
   284→        label: draft.label,
   285→        description: draft.description,
   286→        placeholder: draft.placeholder,
   287→        help_text: draft.help_text,
   288→        is_required: draft.is_required,
   289→        width: draft.width,
   290→        config: draft.config,
   291→      })
   292→      .eq("id", fieldId);
   293→    setSavingFieldId(null);
   294→    if (error) {
   295→      toast.error("Could not save field.", { description: error.message });
   296→      return false;
   297→    }
   298→    setFields((prev) => prev.map((f) => (f.id === fieldId ? { ...f, ...draft } : f)));
   299→    toast.success("Field saved.");
   300→    setEditingFieldId(null);
   301→    return true;
   302→  }
   303→
   304→  async function deleteField(field: FormField) {
   305→    if (
   306→      !confirm(
   307→        `Delete field "${field.label}"? Answers already collected for this field will also be removed. This cannot be undone.`,
   308→      )
   309→    ) {
   310→      return;
   311→    }
   312→    setDeletingFieldId(field.id);
   313→    const { error } = await supabaseBrowser.from("form_fields").delete().eq("id", field.id);
   314→    setDeletingFieldId(null);
   315→    if (error) {
   316→      toast.error("Could not delete field.", { description: error.message });
   317→      return;
   318→    }
   319→    setFields((prev) => prev.filter((f) => f.id !== field.id));
   320→    if (editingFieldId === field.id) setEditingFieldId(null);
   321→    toast.success("Field deleted.");
   322→  }
   323→
   324→  /**
   325→   * Persist a new field order. Only rows whose sort_order actually changed
   326→   * are written (typically 1–3 rows per move) — one batch of updates AFTER
   327→   * the drop, never during dragging.
   328→   */
   329→  async function persistOrder(nextFields: FormField[], previousFields: FormField[]) {
   330→    setReorderError(null);
   331→    setFields(nextFields);
   332→
   333→    const updates: { id: string; sort_order: number }[] = [];
   334→    nextFields.forEach((f, idx) => {
   335→      if (previousFields.find((p) => p.id === f.id)?.sort_order !== idx) {
   336→        updates.push({ id: f.id, sort_order: idx });
   337→      }
   338→    });
   339→    if (updates.length === 0) return;
   340→
   341→    try {
   342→      // Fire the (typically 1–3) row updates as ONE parallel batch —
   343→      // every row is disjoint, so ordering between them is irrelevant.
   344→      // Previously these were sequential awaits: N network round-trips.
   345→      // NOTE: this is still N separate UPDATE statements, not one
   346→      // transaction — a mid-batch failure can leave a partial order in
   347→      // the DB. True atomicity would require a reorder RPC (new function
   348→      // via a future migration); that is intentionally NOT invented here.
   349→      // Instead, on failure we best-effort restore the previous order.
   350→      const results = await Promise.all(
   351→        updates.map((u) =>
   352→          supabaseBrowser
   353→            .from("form_fields")
   354→            .update({ sort_order: u.sort_order })
   355→            .eq("id", u.id),
   356→        ),
   357→      );
   358→      const failed = results.find((r) => r.error);
   359→      if (failed?.error) throw failed.error;
   360→      toast.success("Field order saved.");
   361→    } catch (e) {
   362→      // Best-effort rollback: write the previous sort_orders back so the
   363→      // database matches the restored UI below. Rollback errors are
   364→      // logged but never mask the original failure.
   365→      try {
   366→        await Promise.all(
   367→          previousFields.map((p) =>
   368→            supabaseBrowser
   369→              .from("form_fields")
   370→              .update({ sort_order: p.sort_order })
   371→              .eq("id", p.id),
   372→          ),
   373→        );
   374→      } catch (rollbackErr) {
   375→        console.warn("[builder] order rollback failed:", rollbackErr);
   376→      }
   377→      setFields(previousFields);
   378→      setReorderError(
   379→        "Could not save the new order. Reverted to the last saved order.",
   380→      );
   381→      toast.error("Could not save field order.", {
   382→        description: e instanceof Error ? e.message : "Please try again.",
   383→      });
   384→    }
   385→  }
   386→
   387→  /* ---------------- Drag and drop ---------------- */
   388→
   389→  const sensors = useSensors(
   390→    useSensor(PointerSensor, { activationConstraint: { distance: 8 } }),
   391→    useSensor(TouchSensor, { activationConstraint: { delay: 180, tolerance: 8 } }),
   392→  );
   393→
   394→  function handleDragEnd(event: DragEndEvent) {
   395→    const { active, over } = event;
   396→    if (!over || active.id === over.id) return;
   397→    const oldIndex = fields.findIndex((f) => f.id === active.id);
   398→    const newIndex = fields.findIndex((f) => f.id === over.id);
   399→    if (oldIndex === -1 || newIndex === -1) return;
   400→    const next = arrayMove(fields, oldIndex, newIndex);
   401→    void persistOrder(next, fields);
   402→  }
   403→
   404→  function moveField(fieldId: string, direction: -1 | 1) {
   405→    const index = fields.findIndex((f) => f.id === fieldId);
   406→    const target = index + direction;
   407→    if (index === -1 || target < 0 || target >= fields.length) return;
   408→    const next = arrayMove(fields, index, target);
   409→    void persistOrder(next, fields);
   410→  }
   411→
   412→  /* ---------------- Render ---------------- */
   413→
   414→  if (loading) return <LoadingState title="Loading form…" />;
   415→  if (loadError) return <ErrorState title="Couldn't load form" description={loadError} />;
   416→  if (!form) {
   417→    return (
   418→      <EmptyState
   419→        title="Form not found"
   420→        description="This form may have been deleted, or you don't have access."
   421→        icon={<FileText className="h-7 w-7" />}
   422→        action={{ label: "Back to forms", href: "/dashboard/forms/" }}
   423→      />
   424→    );
   425→  }
   426→
   427→  const statusColor =
   428→    form.status === "published"
   429→      ? "var(--memphis-mint)"
   430→      : form.status === "paused"
   431→        ? "var(--memphis-sun)"
   432→        : form.status === "archived"
   433→          ? "var(--muted-foreground)"
   434→          : "var(--memphis-coral)";
   435→
   436→  return (
   437→    <div className="space-y-6">
   438→      {/* Back link */}
   439→      <div>
   440→        <Link
   441→          href="/dashboard/forms/"
   442→          className="inline-flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground"
   443→        >
   444→          <ArrowLeft className="h-3.5 w-3.5" />
   445→          Back to forms
   446→        </Link>
   447→      </div>
   448→
   449→      {/* Form header */}
   450→      <div className="relative overflow-hidden rounded-2xl border-2 border-foreground/10 bg-surface p-5 sm:p-6">
   451→        <GeometricCircle color="coral" size={32} className="-top-3 -right-3 opacity-80" />
   452→        <div className="relative flex flex-wrap items-start justify-between gap-3">
   453→          <div className="flex items-center gap-3">
   454→            <span
   455→              className="inline-block h-3 w-3 rounded-full"
   456→              style={{ backgroundColor: statusColor }}
   457→              aria-label={form.status}
   458→            />
   459→            <div>
   460→              <h1 className="font-display text-2xl font-bold tracking-tight sm:text-3xl">
   461→                {form.name}
   462→              </h1>
   463→              <p className="text-xs text-muted-foreground">
   464→                Created {new Date(form.created_at).toLocaleDateString()} · Status:{" "}
   465→                <span className="capitalize">{form.status}</span> ·{" "}
   466→                {fields.length} field{fields.length === 1 ? "" : "s"}
   467→              </p>
   468→            </div>
   469→          </div>
   470→          <Button
   471→            variant="outline"
   472→            size="sm"
   473→            onClick={deleteForm}
   474→            disabled={deleting}
   475→            className="border-destructive/30 text-destructive hover:bg-destructive/10"
   476→          >
   477→            <Trash2 className="h-3.5 w-3.5" />
   478→            {deleting ? "Deleting…" : "Delete form"}
   479→          </Button>
   480→        </div>
   481→      </div>
   482→
   483→      {/* Edit form details */}
   484→      <section className="rounded-2xl border-2 border-foreground/10 bg-surface p-5 sm:p-6">
   485→        <h2 className="font-display text-lg font-bold">Form details</h2>
   486→        <p className="mb-4 text-xs text-muted-foreground">
   487→          Edit the name and description. Saved directly to the database.
   488→        </p>
   489→        <div className="space-y-4">
   490→          <div className="space-y-2">
   491→            <Label htmlFor="name">Name</Label>
   492→            <Input
   493→              id="name"
   494→              value={name}
   495→              onChange={(e) => setName(e.target.value)}
   496→              disabled={savingForm}
   497→              className="h-11"
   498→            />
   499→          </div>
   500→          <div className="space-y-2">
   501→            <Label htmlFor="description">Description</Label>
   502→            <Textarea
   503→              id="description"
   504→              value={description}
   505→              onChange={(e) => setDescription(e.target.value)}
   506→              rows={3}
   507→              disabled={savingForm}
   508→              placeholder="What is this form for?"
   509→            />
   510→          </div>
   511→        </div>
   512→        <div className="mt-4 flex justify-end">
   513→          <Button onClick={saveForm} variant="memphis-coral" disabled={savingForm}>
   514→            {savingForm ? "Saving…" : "Save changes"}
   515→          </Button>
   516→        </div>
   517→      </section>
   518→
   519→      {/* Builder */}
   520→      <section className="rounded-2xl border-2 border-foreground/10 bg-surface p-5 sm:p-6">
   521→        <div className="mb-4">
   522→          <h2 className="font-display text-lg font-bold">Fields</h2>
   523→          <p className="text-xs text-muted-foreground">
   524→            Add, edit, reorder and delete fields. Every change is saved to
   525→            Supabase — drag to reorder, or use the arrow buttons.
   526→          </p>
   527→        </div>
   528→
   529→        {reorderError && (
   530→          <p role="alert" className="mb-3 rounded-lg bg-destructive/10 p-2.5 text-xs font-medium text-destructive">
   531→            {reorderError}
   532→          </p>
   533→        )}
   534→
   535→        {fields.length === 0 ? (
   536→          <EmptyState
   537→            title="No fields yet"
   538→            description="Add your first field below. Fields are stored as normalized rows — add, edit, reorder and delete are all real database operations."
   539→            icon={<FileText className="h-6 w-6" />}
   540→          />
   541→        ) : (
   542→          <DndContext
   543→            sensors={sensors}
   544→            collisionDetection={closestCenter}
   545→            onDragEnd={handleDragEnd}
   546→          >
   547→            <SortableContext items={fieldIds} strategy={verticalListSortingStrategy}>
   548→              <ul className="space-y-2">
   549→                {fields.map((field, index) => (
   550→                  <SortableFieldRow
   551→                    key={field.id}
   552→                    field={field}
   553→                    index={index}
   554→                    total={fields.length}
   555→                    editing={editingFieldId === field.id}
   556→                    saving={savingFieldId === field.id}
   557→                    deleting={deletingFieldId === field.id}
   558→                    disabled={editingFieldId !== null || deletingFieldId !== null}
   559→                    onEdit={() =>
   560→                      setEditingFieldId(editingFieldId === field.id ? null : field.id)
   561→                    }
   562→                    onSave={(draft) => saveField(field.id, draft)}
   563→                    onCancel={() => setEditingFieldId(null)}
   564→                    onDelete={() => deleteField(field)}
   565→                    onMove={(dir) => moveField(field.id, dir)}
   566→                  />
   567→                ))}
   568→              </ul>
   569→            </SortableContext>
   570→          </DndContext>
   571→        )}
   572→
   573→        {/* Add field palette */}
   574→        <div className="mt-6 space-y-3 border-t border-foreground/10 pt-5">
   575→          <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   576→            Add a field
   577→          </p>
   578→          {(
   579→            [
   580→              ["Basic", FIELD_TYPES_BY_GROUP.basic],
   581→              ["Choice", FIELD_TYPES_BY_GROUP.choice],

   620→          </p>

   631→function SortableFieldRow({
   632→  field,
   633→  index,
   634→  total,
   635→  editing,
   636→  saving,
   637→  deleting,
   638→  disabled,
   639→  onEdit,
   640→  onSave,
   641→  onCancel,
   642→  onDelete,
   643→  onMove,
   644→}: {
   645→  field: FormField;
   646→  index: number;
   647→  total: number;
   648→  editing: boolean;
   649→  saving: boolean;
   650→  deleting: boolean;
   651→  disabled: boolean;
   652→  onEdit: () => void;
   653→  onSave: (draft: {
   654→    label: string;
   655→    description: string | null;
   656→    placeholder: string | null;
   657→    help_text: string | null;
   658→    is_required: boolean;
   659→    width: number;
   660→    config: Record<string, unknown>;
   661→  }) => Promise<boolean>;
   662→  onCancel: () => void;
   663→  onDelete: () => void;
   664→  onMove: (direction: -1 | 1) => void;
   665→}) {
   666→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } =
   667→    useSortable({ id: field.id });
   668→
   669→  const style = {
   670→    transform: CSS.Transform.toString(transform),
   671→    transition,
   672→  };
   673→
   674→  return (
   675→    <li
   676→      ref={setNodeRef}
   677→      style={style}
   678→      className={cn(
   679→        "rounded-xl border border-foreground/10 bg-background",
   680→        isDragging && "z-10 shadow-lg",
   681→      )}
   682→    >
   683→      <div className="flex flex-wrap items-center gap-2 p-3 sm:flex-nowrap">
   684→        {/* Drag handle */}
   685→        <button
   686→          type="button"
   687→          className="flex h-9 w-6 shrink-0 cursor-grab touch-none items-center justify-center rounded text-muted-foreground/60 hover:text-foreground disabled:opacity-40"
   688→          disabled={disabled}
   689→          {...attributes}
   690→          {...listeners}
   691→          aria-label={`Reorder ${field.label} (drag or use arrow buttons)`}
   692→        >
   693→          <GripVertical className="h-4 w-4" />
   694→        </button>
   695→
   696→        {/* Index + label + key */}
   697→        <span className="w-6 shrink-0 font-display text-xs text-muted-foreground">
   698→          {index + 1}
   699→        </span>
   700→        <div className="min-w-0 flex-1">
   701→          <p className="truncate text-sm font-semibold text-foreground">
   702→            {field.label}
   703→            {field.is_required && (
   704→              <span
   705→                className="ml-1.5 inline-block h-2 w-2 rounded-full bg-[color:var(--memphis-coral)] align-middle"
   706→                aria-label="Required"
   707→              />
   708→            )}
   709→          </p>
   710→          <p className="truncate font-mono text-[11px] text-muted-foreground">
   711→            {field.field_key}
   712→          </p>
   713→        </div>
   714→
   715→        {/* Type badge */}
   716→        <span className="shrink-0 rounded-md bg-muted px-2 py-0.5 text-[11px] font-medium text-foreground/80">
   717→          {fieldLabel(field.field_type)}
   718→        </span>
   719→
   720→        {/* Width badge */}
   721→        <span className="hidden shrink-0 text-[11px] text-muted-foreground sm:inline">
   722→          w:{field.width}
   723→        </span>
   724→
   725→        {/* Move up/down — always available (mobile + keyboard friendly) */}
   726→        <div className="flex shrink-0 items-center gap-0.5">
   727→          <Button
   728→            variant="ghost"
   729→            size="icon-sm"
   730→            onClick={() => onMove(-1)}
   731→            disabled={index === 0 || disabled}
   732→            aria-label={`Move ${field.label} up`}
   733→          >
   734→            <ChevronUp className="h-4 w-4" />
   735→          </Button>
   736→          <Button
   737→            variant="ghost"
   738→            size="icon-sm"
   739→            onClick={() => onMove(1)}
   740→            disabled={index === total - 1 || disabled}
   741→            aria-label={`Move ${field.label} down`}
   742→          >
   743→            <ChevronDown className="h-4 w-4" />
   744→          </Button>
   745→        </div>
   746→
   747→        {/* Edit toggle */}
   748→        <Button
   749→          variant="ghost"
   750→          size="icon-sm"
   751→          onClick={onEdit}
   752→          disabled={deleting || (disabled && !editing)}
   753→          aria-label={editing ? `Close editor for ${field.label}` : `Edit ${field.label}`}
   754→          aria-expanded={editing}
   755→        >
   756→          {editing ? <ChevronIcon className="h-4 w-4" /> : <Pencil className="h-4 w-4" />}
   757→        </Button>
   758→
   759→        {/* Delete */}
   760→        <Button
   761→          variant="ghost"
   762→          size="icon-sm"
   763→          onClick={onDelete}
   764→          disabled={deleting || disabled}
   765→          aria-label={`Delete ${field.label}`}
   766→          className="text-muted-foreground hover:bg-destructive/10 hover:text-destructive"
   767→        >
   768→          {deleting ? "…" : <Trash2 className="h-4 w-4" />}
   769→        </Button>
   770→      </div>
   771→
   772→      {/* Expanded editor */}
   773→      {editing && (
   774→        <div className="px-3 pb-3">
   775→          <FieldEditor
   776→            field={field}
   777→            saving={saving}
   778→            onSave={onSave}
   779→            onCancel={onCancel}
   780→          />
   781→        </div>
   782→      )}
   783→    </li>
   784→  );
   785→}
