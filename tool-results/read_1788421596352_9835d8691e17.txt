     3→     3→/**
     4→     4→ * FormNull — Form Builder (Phase 2)
     5→     5→ * =====================================================================
     6→     6→ * The manual form builder. THE SOURCE OF TRUTH for form structure.
     7→     7→ *
     8→     8→ * Real Supabase persistence:
     9→     9→ *   - Form load:      SELECT forms + form_fields (sorted by sort_order)
    10→    10→ *   - Form save:      UPDATE forms (name, description)
    11→    11→ *   - Form delete:    DELETE forms (cascades fields + submissions via FK)
    12→    12→ *   - Add field:      INSERT form_fields (immediate, sort_order = max+1)
    13→    13→ *   - Edit field:     UPDATE form_fields (explicit Save per field)
    14→    14→ *   - Delete field:   DELETE form_fields (cascades submission_values via FK)
    15→    15→ *   - Reorder:        UPDATE sort_order for CHANGED rows only, once per
    16→    16→ *                     drop / button press (no writes during dragging)
    17→    17→ *
    18→    18→ * No fake success: every mutation awaits the real Supabase response and
    19→    19→ * only updates UI state after the database confirms. Failures surface
    20→    20→ * real errors and preserve unsaved local work.
    21→    21→ *
    22→    22→ * RLS: all queries run as the authenticated user — readers must be
    23→    23→ * workspace members, writers editors+. An unauthorized form id simply
    24→    24→ * returns no row ("Form not found").
    25→    25→ */

    83→    83→function slugify(s: string): string {
    84→    84→  return (
    85→    85→    s
    86→    86→      .toLowerCase()
    87→    87→      .trim()
    88→    88→      .replace(/[^a-z0-9]+/g, "_")
    89→    89→      .replace(/^_+|_+$/g, "")
    90→    90→      .slice(0, 50) || "field"
    91→    91→  );
    92→    92→}
    93→    93→
    94→    94→function uniqueKey(label: string, existing: string[]): string {
    95→    95→  const base = slugify(label);
    96→    96→  if (!existing.includes(base)) return base;
    97→    97→  let i = 2;
    98→    98→  while (existing.includes(`${base}_${i}`)) i += 1;
    99→    99→  return `${base}_${i}`;
   100→   100→}

   106→   106→export function FormDetail({ formId }: { formId: string }) {
   107→   107→  const router = useRouter();
   108→   108→  const { user } = useAuth();
   109→   109→  const [form, setForm] = useState<FormRow | null>(null);
   110→   110→  const [fields, setFields] = useState<FormField[]>([]);
   111→   111→  const [loading, setLoading] = useState(true);
   112→   112→  const [loadError, setLoadError] = useState<string | null>(null);
   113→   113→
   114→   114→  // Form-level editing
   115→   115→  const [name, setName] = useState("");
   116→   116→  const [description, setDescription] = useState("");
   117→   117→  const [savingForm, setSavingForm] = useState(false);
   118→   118→  const [deleting, setDeleting] = useState(false);
   119→   119→
   120→   120→  // Field-level editing
   121→   121→  const [editingFieldId, setEditingFieldId] = useState<string | null>(null);
   122→   122→  const [savingFieldId, setSavingFieldId] = useState<string | null>(null);
   123→   123→  const [addingField, setAddingField] = useState<string | null>(null);
   124→   124→  const [deletingFieldId, setDeletingFieldId] = useState<string | null>(null);
   125→   125→  const [reorderError, setReorderError] = useState<string | null>(null);
   126→   126→
   127→   127→  const fieldIds = useMemo(() => fields.map((f) => f.id), [fields]);
   128→   128→
   129→   129→  const load = useCallback(async () => {
   130→   130→    setLoading(true);
   131→   131→    setLoadError(null);
   132→   132→    try {
   133→   133→      const [formRes, fieldsRes] = await Promise.all([
   134→   134→        supabaseBrowser.from("forms").select("*").eq("id", formId).maybeSingle(),
   135→   135→        supabaseBrowser
   136→   136→          .from("form_fields")
   137→   137→          .select("*")
   138→   138→          .eq("form_id", formId)
   139→   139→          .order("sort_order", { ascending: true }),
   140→   140→      ]);
   141→   141→
   142→   142→      if (formRes.error) throw formRes.error;
   143→   143→      if (!formRes.data) {
   144→   144→        setForm(null);
   145→   145→        setLoadError("Form not found, or you don't have access to it.");
   146→   146→        setLoading(false);
   147→   147→        return;
   148→   148→      }
   149→   149→      setForm(formRes.data);
   150→   150→      setName(formRes.data.name);
   151→   151→      setDescription(formRes.data.description ?? "");
   152→   152→
   153→   153→      if (fieldsRes.error) throw fieldsRes.error;
   154→   154→      setFields((fieldsRes.data as FormField[]) ?? []);
   155→   155→    } catch (e) {
   156→   156→      const msg = e instanceof Error ? e.message : "Failed to load form.";
   157→   157→      setLoadError(msg);
   158→   158→    } finally {
   159→   159→      setLoading(false);
   160→   160→    }
   161→   161→  }, [formId]);
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    load();
   165→   165→  }, [load]);

   169→   169→  async function saveForm() {
   170→   170→    if (!form) return;
   171→   171→    if (!name.trim()) {
   172→   172→      toast.error("Form name cannot be empty.");
   173→   173→      return;
   174→   174→    }
   175→   175→    // updated_by must be the AUTHENTICATED current user — never a
   176→   176→    // fabricated or inherited id (previously this wrote created_by,
   177→   177→    // which misattributes edits when a workspace admin saves the form).
   178→   178→    if (!user) {
   179→   179→      toast.error("Your session has expired. Please sign in again.");
   180→   180→      return;
   181→   181→    }
   182→   182→    setSavingForm(true);
   183→   183→    const { error } = await supabaseBrowser
   184→   184→      .from("forms")
   185→   185→      .update({
   186→   186→        name: name.trim(),
   187→   187→        description: description.trim() || null,
   188→   188→        updated_by: user.id,
   189→   189→      })
   190→   190→      .eq("id", form.id);
   191→   191→    setSavingForm(false);
   192→   192→    if (error) {
   193→   193→      toast.error("Could not save form.", { description: error.message });
   194→   194→      return;
   195→   195→    }
   196→   196→    toast.success("Form updated.");
   197→   197→    setForm({
   198→   198→      ...form,
   199→   199→      name: name.trim(),
   200→   200→      description: description.trim() || null,
   201→   201→      updated_by: user.id,
   202→   202→    });
   203→   203→  }

   205→   205→  async function deleteForm() {
   206→   206→    if (!form) return;
   207→   207→    if (!confirm(`Delete "${form.name}"? All fields and submissions will be permanently removed. This cannot be undone.`)) {
   208→   208→      return;
   209→   209→    }
   210→   210→    setDeleting(true);
   211→   211→    const { error } = await supabaseBrowser.from("forms").delete().eq("id", form.id);
   212→   212→    setDeleting(false);
   213→   213→    if (error) {
   214→   214→      toast.error("Could not delete form.", { description: error.message });
   215→   215→      return;
   216→   216→    }
   217→   217→    toast.success("Form deleted.");
   218→   218→    router.push("/dashboard/forms/");

   220→   220→  }

   224→   224→  async function addField(type: FieldType) {
   225→   225→    if (!form) return;
   226→   226→    const meta = { type, label: fieldLabel(type) };
   227→   227→    setAddingField(type);
   228→   228→    try {
   229→   229→      const existingKeys = fields.map((f) => f.field_key);
   230→   230→      const key = uniqueKey(meta.label, existingKeys);
   231→   231→      const nextOrder = fields.length > 0 ? Math.max(...fields.map((f) => f.sort_order)) + 1 : 0;
   232→   232→
   233→   233→      // Type-appropriate default config from the CENTRALIZED registry
   234→   234→      // (selects start with valid options, ratings with max=5).
   235→   235→      const defaultConfig = defaultConfigForType(type);
   236→   236→
   237→   237→      const { data, error } = await supabaseBrowser
   238→   238→        .from("form_fields")
   239→   239→        .insert({
   240→   240→          form_id: form.id,
   241→   241→          field_key: key,
   242→   242→          field_type: type,
   243→   243→          label: meta.label,
   244→   244→          is_required: false,
   245→   245→          sort_order: nextOrder,
   246→   246→          width: 12,
   247→   247→          config: defaultConfig,
   248→   248→        })
   249→   249→        .select()
   250→   250→        .single();
   251→   251→
   252→   252→      if (error) throw error;
   253→   253→      if (!data) throw new Error("Field creation returned no row.");
   254→   254→
   255→   255→      // Real row confirmed by the database — now update the UI.
   256→   256→      setFields((prev) => [...prev, data as FormField]);
   257→   257→      setEditingFieldId(data.id);
   258→   258→      toast.success(`${meta.label} field added.`);
   259→   259→    } catch (e) {
   260→   260→      toast.error("Could not add field.", {
   261→   261→        description: e instanceof Error ? e.message : "Please try again.",
   262→   262→      });
   263→   263→    } finally {
   264→   264→      setAddingField(null);
   265→   265→    }
   266→   266→  }

   304→   304→  async function deleteField(field: FormField) {
   305→   305→    if (
   306→   306→      !confirm(
   307→   307→        `Delete field "${field.label}"? Answers already collected for this field will also be removed. This cannot be undone.`,
   308→   308→      )
   309→   309→    ) {
   310→   310→      return;
   311→   311→    }
   312→   312→    setDeletingFieldId(field.id);
   313→   313→    const { error } = await supabaseBrowser.from("form_fields").delete().eq("id", field.id);
   314→   314→    setDeletingFieldId(null);
   315→   315→    if (error) {
   316→   316→      toast.error("Could not delete field.", { description: error.message });
   317→   317→      return;
   318→   318→    }
   319→   319→    setFields((prev) => prev.filter((f) => f.id !== field.id));
   320→   320→    if (editingFieldId === field.id) setEditingFieldId(null);
   321→   321→    toast.success("Field deleted.");
   322→   322→  }
   323→   323→
   324→   324→  /**
   325→   325→   * Persist a new field order. Only rows whose sort_order actually changed
   326→   326→   * are written (typically 1–3 rows per move) — one batch of updates AFTER
   327→   327→   * the drop, never during dragging.
   328→   328→   */
   329→   329→  async function persistOrder(nextFields: FormField[], previousFields: FormField[]) {
   330→   330→    setReorderError(null);
   331→   331→    setFields(nextFields);
   332→   332→
   333→   333→    const updates: { id: string; sort_order: number }[] = [];
   334→   334→    nextFields.forEach((f, idx) => {
   335→   335→      if (previousFields.find((p) => p.id === f.id)?.sort_order !== idx) {
   336→   336→        updates.push({ id: f.id, sort_order: idx });
   337→   337→      }
   338→   338→    });
   339→   339→    if (updates.length === 0) return;
   340→   340→
   341→   341→    try {
   342→   342→      // Fire the (typically 1–3) row updates as ONE parallel batch —
   343→   343→      // every row is disjoint, so ordering between them is irrelevant.
   344→   344→      // Previously these were sequential awaits: N network round-trips.
   345→   345→      // NOTE: this is still N separate UPDATE statements, not one
   346→   346→      // transaction — a mid-batch failure can leave a partial order in
   347→   347→      // the DB. True atomicity would require a reorder RPC (new function
   348→   348→      // via a future migration); that is intentionally NOT invented here.
   349→   349→      // Instead, on failure we best-effort restore the previous order.
   350→   350→      const results = await Promise.all(
   351→   351→        updates.map((u) =>
   352→   352→          supabaseBrowser
   353→   353→            .from("form_fields")
   354→   354→            .update({ sort_order: u.sort_order })
   355→   355→            .eq("id", u.id),
   356→   356→        ),
   357→   357→      );
   358→   358→      const failed = results.find((r) => r.error);
   359→   359→      if (failed?.error) throw failed.error;
   360→   360→      toast.success("Field order saved.");
   361→   361→    } catch (e) {
   362→   362→      // Best-effort rollback: write the previous sort_orders back so the
   363→   363→      // database matches the restored UI below. Rollback errors are
   364→   364→      // logged but never mask the original failure.
   365→   365→      try {
   366→   366→        await Promise.all(
   367→   367→          previousFields.map((p) =>
   368→   368→            supabaseBrowser
   369→   369→              .from("form_fields")
   370→   370→              .update({ sort_order: p.sort_order })
   371→   371→              .eq("id", p.id),
   372→   372→          ),
   373→   373→        );
   374→   374→      } catch (rollbackErr) {
   375→   375→        console.warn("[builder] order rollback failed:", rollbackErr);
   376→   376→      }
   377→   377→      setFields(previousFields);
   378→   378→      setReorderError(
   379→   379→        "Could not save the new order. Reverted to the last saved order.",
   380→   380→      );
   381→   381→      toast.error("Could not save field order.", {
   382→   382→        description: e instanceof Error ? e.message : "Please try again.",
   383→   383→      });
   384→   384→    }
   385→   385→  }
   386→   386→
   387→   387→  /* ---------------- Drag and drop ---------------- */
   388→   388→
   389→   389→  const sensors = useSensors(
   390→   390→    useSensor(PointerSensor, { activationConstraint: { distance: 8 } }),
   391→   391→    useSensor(TouchSensor, { activationConstraint: { delay: 180, tolerance: 8 } }),
   392→   392→  );
   393→   393→
   394→   394→  function handleDragEnd(event: DragEndEvent) {
   395→   395→    const { active, over } = event;
   396→   396→    if (!over || active.id === over.id) return;
   397→   397→    const oldIndex = fields.findIndex((f) => f.id === active.id);
   398→   398→    const newIndex = fields.findIndex((f) => f.id === over.id);
   399→   399→    if (oldIndex === -1 || newIndex === -1) return;
   400→   400→    const next = arrayMove(fields, oldIndex, newIndex);
   401→   401→    void persistOrder(next, fields);
   402→   402→  }
   403→   403→
   404→   404→  function moveField(fieldId: string, direction: -1 | 1) {
   405→   405→    const index = fields.findIndex((f) => f.id === fieldId);
   406→   406→    const target = index + direction;
   407→   407→    if (index === -1 || target < 0 || target >= fields.length) return;
   408→   408→    const next = arrayMove(fields, index, target);
   409→   409→    void persistOrder(next, fields);
   410→   410→  }

   422→   422→        action={{ label: "Back to forms", href: "/dashboard/forms/" }}
   423→   423→      />
   424→   424→    );
   425→   425→  }
   426→   426→

   429→   429→      ? "var(--memphis-mint)"

   435→   435→
   436→   436→  return (
   437→   437→    <div className="space-y-6">
   438→   438→      {/* Back link */}
   439→   439→      <div>

   443→   443→        >
   444→   444→          <ArrowLeft className="h-3.5 w-3.5" />
   445→   445→          Back to forms
   446→   446→        </Link>
   447→   447→      </div>
   448→   448→
   449→   449→      {/* Form header */}
   450→   450→      <div className="relative overflow-hidden rounded-2xl border-2 border-foreground/10 bg-surface p-5 sm:p-6">
   451→   451→        <GeometricCircle color="coral" size={32} className="-top-3 -right-3 opacity-80" />
   452→   452→        <div className="relative flex flex-wrap items-start justify-between gap-3">
   453→   453→          <div className="flex items-center gap-3">
   454→   454→            <span
   455→   455→              className="inline-block h-3 w-3 rounded-full"
   456→   456→              style={{ backgroundColor: statusColor }}
   457→   457→              aria-label={form.status}
   458→   458→            />
   459→   459→            <div>
   460→   460→              <h1 className="font-display text-2xl font-bold tracking-tight sm:text-3xl">
   461→   461→                {form.name}
   462→   462→              </h1>
   463→   463→              <p className="text-xs text-muted-foreground">
   464→   464→                Created {new Date(form.created_at).toLocaleDateString()} · Status:{" "}
   465→   465→                <span className="capitalize">{form.status}</span> ·{" "}
   466→   466→                {fields.length} field{fields.length === 1 ? "" : "s"}
   467→   467→              </p>
   468→   468→            </div>
   469→   469→          </div>
   470→   470→          <Button
   471→   471→            variant="outline"
   472→   472→            size="sm"
   473→   473→            onClick={deleteForm}
   474→   474→            disabled={deleting}
   475→   475→            className="border-destructive/30 text-destructive hover:bg-destructive/10"
   476→   476→          >
   477→   477→            <Trash2 className="h-3.5 w-3.5" />
   478→   478→            {deleting ? "Deleting…" : "Delete form"}
   479→   479→          </Button>
   480→   480→        </div>
