     4→ * FormNull — Form Builder (Phase 3)
     5→ * =====================================================================
     6→ * THE manual form builder — the product's core surface. Professional
     7→ * 3-pane layout, Memphis-designed:
     8→ *
     9→ *   ┌────────────────────────────────────────────────────────────┐
    10→ *   │ toolbar: back · status · name · save-state · preview ·     │
    11→ *   │          publish · share · more                            │
    12→ *   ├───────────┬───────────────────────────┬────────────────────┤
    13→ *   │ field     │ live form canvas          │ properties         │
    14→ *   │ library   │ (what respondents see)    │ (form / field)     │
    15→ *   └───────────┴───────────────────────────┴────────────────────┘
    16→ *
    17→ * Tablet/mobile (< lg): canvas full-width; the field library and the
    18→ * properties panel become Sheets. Nothing is unreachable.
    19→ *
    20→ * Real Supabase persistence (unchanged contracts from Phase 2A):
    21→ *   - Form load:      SELECT forms + form_fields (sorted by sort_order)
    22→ *   - Form save:      UPDATE forms (name, description, settings)
    23→ *   - Form delete:    DELETE forms (cascades fields + submissions)
    24→ *   - Add field:      INSERT form_fields (immediate, sort_order = max+1)
    25→ *   - Edit field:     UPDATE form_fields (explicit Save per field)
    26→ *   - Duplicate:      INSERT form_fields (config copy, unique key)
    27→ *   - Delete field:   DELETE form_fields (006 keeps collected answers:
    28→ *                     submission_values.field_id → ON DELETE SET NULL)
    29→ *   - Reorder:        UPDATE sort_order for changed rows only, once
    30→ *                     per drop / button press, with rollback.
    31→ *   - Publish:        RPC publish_form (migration 006 — immutable
    32→ *                     version snapshot + public link flip).
    33→ *
    34→ * No fake success: every mutation awaits the real Supabase response
    35→ * and only updates UI state after the database confirms. Failures
    36→ * surface real errors and preserve unsaved local work — selection
    37→ * changes and unload are guarded while edits are pending.
    38→ *
    39→ * RLS: all queries run as the authenticated user — readers must be
    40→ * workspace members, writers editors+. An unauthorized form id simply
    41→ * returns no row ("Form not found"). publish_form re-checks editor
    42→ * rights server-side (SECURITY DEFINER) — the button being visible
    43→ * is never the permission boundary.
    44→ */

   136→function slugify(s: string): string {
   137→  return (
   138→    s
   139→      .toLowerCase()
   140→      .trim()
   141→      .replace(/[^a-z0-9]+/g, "_")
   142→      .replace(/^_+|_+$/g, "")
   143→      .slice(0, 50) || "field"
   144→  );
   145→}
   146→
   147→function uniqueKey(base: string, existing: string[]): string {
   148→  const b = slugify(base);
   149→  if (!existing.includes(b)) return b;
   150→  let i = 2;
   151→  while (existing.includes(`${b}_${i}`)) i += 1;
   152→  return `${b}_${i}`;
   153→}

   373→  async function addField(type: FieldType) {
   374→    if (!form) return;

   376→      toast.error("Field limit reached.", {

   378→      });
   379→      return;
   380→    }
   381→    const meta = { type, label: fieldLabel(type) };
   382→    setAddingField(type);
   383→    try {
   384→      const existingKeys = fields.map((f) => f.field_key);
   385→      const key = uniqueKey(meta.label, existingKeys);
   386→      const nextOrder =
   387→        fields.length > 0 ? Math.max(...fields.map((f) => f.sort_order)) + 1 : 0;
   388→
   389→      // Type-appropriate default config from the CENTRALIZED registry
   390→      // (selects start with valid options, ratings with max=5).
   391→      const defaultConfig = defaultConfigForType(type);
   392→
   393→      const { data, error } = await supabaseBrowser
   394→        .from("form_fields")
   395→        .insert({
   396→          form_id: form.id,
   397→          field_key: key,
   398→          field_type: type,
   399→          label: meta.label,
   400→          is_required: false,
   401→          sort_order: nextOrder,
   402→          width: 12,
   403→          config: defaultConfig,
   404→        })
   405→        .select()
   406→        .single();
   407→
   408→      if (error) throw error;
   409→      if (!data) throw new Error("Field creation returned no row.");
   410→
   411→      // Real row confirmed by the database — now update the UI.
   412→      setFields((prev) => [...prev, data as FormField]);
   413→      selectField(data.id);
   414→      if (!isDesktop) setLibrarySheetOpen(false);

   424→        description: e instanceof Error ? e.message : "Please try again.",

   540→  /**
   541→   * Persist a new field order. Only rows whose sort_order actually changed
   542→   * are written (typically 1–3 rows per move) — one batch of updates AFTER
   543→   * the drop, never during dragging. On failure: best-effort rollback of
   544→   * the previous order (documented limitation: not a single transaction).
   545→   */
   546→  async function persistOrder(nextFields: FormField[], previousFields: FormField[]) {
   547→    setReorderError(null);
   548→    setFields(nextFields);
   549→
   550→    const updates: { id: string; sort_order: number }[] = [];
   551→    nextFields.forEach((f, idx) => {
   552→      if (previousFields.find((p) => p.id === f.id)?.sort_order !== idx) {
   553→        updates.push({ id: f.id, sort_order: idx });
   554→      }
   555→    });
   556→    if (updates.length === 0) return;
   557→
   558→    try {
   559→      const results = await Promise.all(
   560→        updates.map((u) =>
   561→          supabaseBrowser
   562→            .from("form_fields")
   563→            .update({ sort_order: u.sort_order })
   564→            .eq("id", u.id),
   565→        ),
   566→      );
   567→      const failed = results.find((r) => r.error);
   568→      if (failed?.error) throw failed.error;
   569→    } catch (e) {
   570→      try {
   571→        await Promise.all(
   572→          previousFields.map((p) =>
   573→            supabaseBrowser
   574→              .from("form_fields")
   575→              .update({ sort_order: p.sort_order })
   576→              .eq("id", p.id),
   577→          ),
   578→        );
   579→      } catch (rollbackErr) {
   580→        console.warn("[builder] order rollback failed:", rollbackErr);
   581→      }
   582→      setFields(previousFields);
   583→      setReorderError(
   584→        "Could not save the new order. Reverted to the last saved order.",
   585→      );
   586→      toast.error("Could not save field order.", {
   587→        description: e instanceof Error ? e.message : "Please try again.",
   588→      });
   589→    }
   590→  }

   599→  function handleDragEnd(event: DragEndEvent) {
   600→    const { active, over } = event;
   601→    if (!over || active.id === over.id) return;

   603→    const newIndex = fields.findIndex((f) => f.id === over.id);
   604→    if (oldIndex === -1 || newIndex === -1) return;
   605→    const next = arrayMove(fields, oldIndex, newIndex);
   606→    void persistOrder(next, fields);
   607→  }
   608→
   609→  function moveField(fieldId: string, direction: -1 | 1) {
   610→    const index = fields.findIndex((f) => f.id === fieldId);
   611→    const target = index + direction;
   612→    if (index === -1 || target < 0 || target >= fields.length) return;
   613→    const next = arrayMove(fields, index, target);
   614→    void persistOrder(next, fields);
   615→  }

  1121→function SaveStateChip({
  1122→  saving,
  1123→  formDirty,
  1124→  editorDirty,
  1125→}: {
  1126→  saving: boolean;
  1127→  formDirty: boolean;
  1128→  editorDirty: boolean;
  1129→}) {
  1130→  let label: string;
  1131→  let cls: string;
  1132→  let icon = <Check className="h-3 w-3" aria-hidden />;
  1133→  if (saving) {
  1134→    label = "Saving…";
  1135→    cls = "bg-[color:var(--memphis-sun)]/15 text-[color:var(--memphis-sun)]";
  1136→    icon = <Loader2 className="h-3 w-3 animate-spin" aria-hidden />;
  1137→  } else if (editorDirty) {
  1138→    label = "Unsaved field edits";
  1139→    cls = "bg-[color:var(--memphis-coral)]/12 text-[color:var(--memphis-coral)]";
  1140→    icon = <TriangleAlert className="h-3 w-3" aria-hidden />;
  1141→  } else if (formDirty) {
  1142→    label = "Unsaved changes";
  1143→    cls = "bg-[color:var(--memphis-coral)]/12 text-[color:var(--memphis-coral)]";
  1144→    icon = <TriangleAlert className="h-3 w-3" aria-hidden />;
  1145→  } else {
  1146→    label = "All changes saved";
  1147→    cls = "bg-[color:var(--memphis-mint)]/15 text-[color:var(--memphis-mint)]";
