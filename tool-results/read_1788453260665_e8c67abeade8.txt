     1→/**
     2→ * FormNull — Field System 2.0 Registry
     3→ * =====================================================================
     4→ * THE single source of truth for what a field type IS and what it can
     5→ * be configured to do. Consumers:
     6→ *
     7→ *   - field-library.tsx       which types can be ADDED (status=active)
     8→ *   - field-property-editor   which properties exist per type (schema)
     9→ *   - form-renderer.tsx       renders every type, incl. legacy
    10→ *   - publish/public submit   enforced by migration 006 (server side)
    11→ *
    12→ * ── STATUS MODEL ─────────────────────────────────────────────────────
    13→ *   active : offered in the field library, full Field System 2.0
    14→ *            property depth (first 10 types of the rebuild).
    15→ *   legacy : NOT offered for new fields, but existing forms keep
    16→ *            rendering / editing / publishing / submitting them
    17→ *            through this same registry (compatibility layer — no
    18→ *            destructive migration, no data loss).
    19→ *
    20→ * ── HONEST-PROPERTY PRINCIPLE (extends the Phase 3 rule) ────────────
    21→ * Every PropertyDefinition declares where it lives and who enforces it:
    22→ *
    23→ *   target: "column"  → a real form_fields column (002/006 snapshot)
    24→ *   target: "config"  → a key inside form_fields.config (jsonb)
    25→ *
    26→ *   enforcement:
    27→ *     "server"       → submit_public_form (006) validates it on every
    28→ *                      public submission. Fully honest validation.
    29→ *     "client"       → enforced by the shared renderer only (e.g. the
    30→ *                      short_text JS `pattern`, decimal `precision`).
    31→ *                      Documented as a browser-side check; the server
    32→ *                      still applies its own type/length backstop.
    33→ *     "presentation" → changes rendering only (labels, symbols,
    34→ *                      rows). Flows through the 006 snapshot to the
    35→ *                      public form unchanged.
    36→ *
    37→ * No property is declared unless at least one of those is true. No UI
    38→ * control ships without a real consumer.
    39→ *
    40→ * ── EXTENSION CONTRACT (Logic / Calculations / Themes / Integrations) ──
    41→ * Future systems must be able to reason about fields WITHOUT touching
    42→ * the field model. They consume, they do not rewrite:
    43→ *   - `answerType`  : the JSON shape a submitted answer takes — Logic
    44→ *                     conditions and Calculations branch on this.
    45→ *   - `collectsData`: whether the field participates in submissions.
    46→ *   - `publishable` : whether it can enter a published snapshot.
    47→ * A "card / one-question-at-a-time" form mode consumes the exact same
    48→ * field rows — presentation mode is a renderer concern, never a field
    49→ * definition concern.
    50→ */

    77→/** Sections render in this order in the property editor. */
    78→export const PROPERTY_SECTION_ORDER = [
    79→  "general",
    80→  "content",
    81→  "options",
    82→  "validation",
    83→  "appearance",
    84→  "behavior",
    85→] as const;
    86→
    87→export type PropertySection = (typeof PROPERTY_SECTION_ORDER)[number];
    88→
    89→export const SECTION_LABELS: Record<PropertySection, string> = {
    90→  general: "General",
    91→  content: "Content",
    92→  options: "Options",
    93→  validation: "Validation",
    94→  appearance: "Appearance",
    95→  behavior: "Behavior",
    96→};

    98→/** Who guarantees a validation-ish property actually works. */
    99→export type Enforcement = "server" | "client" | "presentation";
   100→
   101→/** Widget used by the generic property editor. */
   102→export type PropertyControl =
   103→  | "text"
   104→  | "textarea"
   105→  | "number"
   106→  | "switch"
   107→  | "width"
   108→  | "select"
   109→  | "options-editor"
   110→  | "default-country";
   111→
   112→export interface PropertyDefinition {
   113→  /** config key (target=config) or column name (target=column). */
   114→  key: string;
   115→  label: string;
   116→  section: PropertySection;
   117→  target: "column" | "config";
   118→  control: PropertyControl;
   119→  enforcement?: Enforcement;
   120→  /** One-line hint under the control. */
   121→  hint?: string;
   122→  placeholder?: string;
   123→  min?: number;
   124→  max?: number;
   125→  step?: number;
   126→  /** Span both grid columns (labels, long inputs). */
   127→  fullWidth?: boolean;
   128→  /** Only render when the existing config already carries the key —
   129→   *  used to keep managing legacy settings honestly instead of
   130→   *  hiding them (e.g. phone minLength from pre-rebuild fields). */
   131→  visibleWhenPresent?: boolean;
   132→  /** Choices for control="select". */
   133→  choices?: { value: string; label: string }[];
   134→}

   150→/**
   151→ * The JSON shape one submitted answer takes. The axis every future
   152→ * Logic / Calculation / Integration ruleset branches on.
   153→ */
   154→export type AnswerType = "string" | "number" | "boolean" | "string[]" | "none";
   155→
   156→export interface FieldTypeDef {
   157→  value: FieldType;
   158→  label: string;
   159→  icon: LucideIcon;
   160→  /** One-line explanation (library + canvas badges). */
   161→  description: string;
   162→  group?: FieldGroup;
   163→  /** active = addable in the library. legacy = compatibility only. */
   164→  status: "active" | "legacy";
   165→  answerType: AnswerType;
   166→  collectsData: boolean;
   167→  publishable: boolean;
   168→  defaultLabel: string;
   169→  /** Config written on field creation (selects need valid options). */
   170→  defaultConfig: () => Record<string, unknown>;
   171→  /** The complete property set — exactly what this type supports. */
   172→  properties: PropertyDefinition[];
   173→  /**
   174→   * Static note shown at the top of the Validation section when the
   175→   * type has built-in (non-configurable) server validation.
   176→   */
   177→  validationNote?: string;
   178→}

   911→export function fieldDef(type: FieldType): FieldTypeDef {
   912→  const def = FIELD_REGISTRY.find((t) => t.value === type);
   913→  if (def) return def;
   914→  // Unreachable while the registry covers the full DB enum; keep the
   915→  // builder honest instead of crashing on future enum values.
   916→  throw new Error(`Unknown field type: ${type}`);
   917→}
   918→
   919→export function fieldDefSafe(type: FieldType): FieldTypeDef | undefined {
   920→  return FIELD_REGISTRY.find((t) => t.value === type);
   921→}

   923→/** Types offered in the field library (active only, in group order). */
   924→export const ADDABLE_FIELD_TYPES: FieldTypeDef[] = FIELD_REGISTRY.filter(
   925→  (t) => t.status === "active",
   926→);
   927→
   928→export const FIELD_TYPES_BY_GROUP: Record<FieldGroup, FieldTypeDef[]> =
   929→  FIELD_GROUPS.reduce(
   930→    (acc, g) => {
   931→      acc[g.key] = ADDABLE_FIELD_TYPES.filter((t) => t.group === g.key);
   932→      return acc;
   933→    },
   934→    {} as Record<FieldGroup, FieldTypeDef[]>,
   935→  );

   938→export const fieldMeta = fieldDefSafe;

   940→  return fieldDefSafe(type)?.label ?? type;

   946→/* ------------------------------------------------------------------ */
   947→/* Config validation — mirrors 006 publish_form's checks so the       */
   948→/* editor blocks invalid configs BEFORE they reach the database       */
   949→/* ------------------------------------------------------------------ */
   950→
   951→export interface ConfigValidation {
   952→  ok: boolean;
   953→  message?: string;
   954→}
   955→
   956→/**
   957→ * A number 006 can round-trip: its numeric regex is
   958→ * ^-?[0-9]{1,10}([.][0-9]{1,6})?$ — JS floats outside that range
   959→ * (1e21, 1e-8) serialize exponentially and would be rejected at
   960→ * publish time. Catch them here, honestly, up front.
   961→ */
   962→function isRepresentableNumber(v: unknown): boolean {
   963→  if (typeof v !== "number" || !Number.isFinite(v)) return false;
   964→  if (Math.abs(v) >= 1e10) return false;
   965→  const s = String(v);
   966→  if (s.includes("e") || s.includes("E")) return false;
   967→  const decimals = s.includes(".") ? s.split(".")[1].length : 0;
   968→  return decimals <= 6;
   969→}
   970→
   971→function intInRange(v: unknown, min: number, max: number): boolean {
   972→  return (
   973→    typeof v === "number" && Number.isInteger(v) && v >= min && v <= max
   974→  );
   975→}
   976→
   977→/** Validate an option list (values + labels) for select-like fields. */
   978→export function validateOptions(
   979→  options: string[],
   980→  labels: Record<string, string>,
   981→): ConfigValidation {
   982→  if (!Array.isArray(options) || options.length === 0) {
   983→    return { ok: false, message: "Add at least one option." };
   984→  }
   985→  if (options.length > MAX_OPTIONS_PER_FIELD) {
   986→    return {
   987→      ok: false,
   988→      message: `At most ${MAX_OPTIONS_PER_FIELD} options are allowed.`,
   989→    };
   990→  }
   991→  const seen = new Set<string>();
   992→  for (const v of options) {
   993→    if (typeof v !== "string" || v.trim().length === 0) {
   994→      return { ok: false, message: "Option values cannot be empty." };
   995→    }
   996→    if (v.length > MAX_OPTION_LEN) {
   997→      return {
   998→        ok: false,
   999→        message: `Option values must be at most ${MAX_OPTION_LEN} characters.`,
  1000→      };
  1001→    }
  1002→    if (seen.has(v)) {
  1003→      return { ok: false, message: "Option values must be unique." };
  1004→    }
  1005→    seen.add(v);
  1006→  }
  1007→  for (const [value, label] of Object.entries(labels)) {
  1008→    if (typeof label !== "string") continue;
  1009→    if (label.length > MAX_PRESENTATION_LEN) {
  1010→      return {
  1011→        ok: false,
  1012→        message: `Option labels must be at most ${MAX_PRESENTATION_LEN} characters.`,
  1013→      };
  1014→    }
  1015→    if (!options.includes(value) && label.trim() !== "") {
  1016→      // Stale label entry — harmless at render (never looked up), but
  1017→      // keep configs clean.
  1018→      return { ok: false, message: "An option label has no matching value." };
  1019→    }
  1020→  }
  1021→  return { ok: true };
  1022→}
  1023→
  1024→/**
  1025→ * Validate a full config object for a type before writing it to the
  1026→ * database. Mirrors publish_form() so nothing valid here can fail
  1027→ * there, and nothing invalid here can silently pass there.
  1028→ */
  1029→export function validateConfig(
  1030→  type: FieldType,
  1031→  config: Record<string, unknown>,
  1032→): ConfigValidation {
  1033→  switch (type) {
  1034→    case "short_text":
  1035→    case "long_text":
  1036→    case "email":
  1037→    case "url":
  1038→    case "phone": {
  1039→      const minL = config.minLength;
  1040→      const maxL = config.maxLength;
  1041→      if (
  1042→        minL != null &&
  1043→        (!intInRange(minL, 0, MAX_LENGTH_CFG))
  1044→      ) {
  1045→        return {
  1046→          ok: false,
  1047→          message: `Min length must be a whole number between 0 and ${MAX_LENGTH_CFG}.`,
  1048→        };
  1049→      }
  1050→      if (maxL != null && !intInRange(maxL, 1, MAX_LENGTH_CFG)) {
  1051→        return {
  1052→          ok: false,
  1053→          message: `Max length must be a whole number between 1 and ${MAX_LENGTH_CFG}.`,
  1054→        };
  1055→      }
  1056→      if (minL != null && maxL != null && Number(minL) > Number(maxL)) {
  1057→        return { ok: false, message: "Min length cannot exceed max length." };
  1058→      }
  1059→      if (type === "short_text" && typeof config.pattern === "string" && config.pattern) {
  1060→        if (config.pattern.length > MAX_PATTERN_LEN) {
  1061→          return {
  1062→            ok: false,
  1063→            message: `Pattern must be at most ${MAX_PATTERN_LEN} characters.`,
  1064→          };
  1065→        }
  1066→        try {
  1067→          new RegExp(config.pattern);
  1068→        } catch {
  1069→          return { ok: false, message: "Pattern is not a valid regular expression." };
  1070→        }
  1071→      }
  1072→      if (type === "phone" && config.defaultCountry != null) {
  1073→        if (typeof config.defaultCountry !== "string" || !/^[A-Z]{2}$/.test(config.defaultCountry)) {
  1074→          return { ok: false, message: "Default country must be a 2-letter code." };
  1075→        }
  1076→      }
  1077→      return { ok: true };
  1078→    }
  1079→
  1080→    case "number":
  1081→    case "decimal":
  1082→    case "scale": {
  1083→      const { min, max, step } = config;
  1084→      if (min != null && !isRepresentableNumber(min)) {
  1085→        return { ok: false, message: "Min must be a number (up to 10 digits, 6 decimals)." };
  1086→      }
  1087→      if (max != null && !isRepresentableNumber(max)) {
  1088→        return { ok: false, message: "Max must be a number (up to 10 digits, 6 decimals)." };
  1089→      }
  1090→      if (step != null && (!isRepresentableNumber(step) || Number(step) <= 0)) {
  1091→        return { ok: false, message: "Step must be greater than 0 (up to 6 decimals)." };
  1092→      }
  1093→      if (min != null && max != null && Number(min) >= Number(max)) {
  1094→        return { ok: false, message: "Min must be less than max." };
  1095→      }
  1096→      if (type === "decimal" && config.precision != null) {
  1097→        if (!intInRange(config.precision, 0, 6)) {
  1098→          return { ok: false, message: "Precision must be a whole number between 0 and 6." };
  1099→        }
  1100→      }
  1101→      if (type === "scale") {
  1102→        for (const k of ["leftLabel", "rightLabel"] as const) {
  1103→          const v = config[k];
  1104→          if (typeof v === "string" && v.length > MAX_END_LABEL_LEN) {
  1105→            return {
  1106→              ok: false,
  1107→              message: `${k === "leftLabel" ? "Left" : "Right"} label must be at most ${MAX_END_LABEL_LEN} characters.`,
  1108→            };
  1109→          }
  1110→        }
  1111→      }
  1112→      return { ok: true };
  1113→    }
  1114→
  1115→    case "single_select":
  1116→    case "multi_select": {
  1117→      const options = Array.isArray(config.options)
  1118→        ? (config.options as unknown[])
  1119→        : [];
  1120→      const labels =
  1121→        config.optionLabels && typeof config.optionLabels === "object"
  1122→          ? (config.optionLabels as Record<string, string>)
  1123→          : {};
  1124→      const allStrings = options.every((o) => typeof o === "string");
  1125→      if (!allStrings) {
  1126→        return { ok: false, message: "Option values must be plain strings." };
  1127→      }
  1128→      return validateOptions(options as string[], labels);
  1129→    }
  1130→
  1131→    case "rating": {
  1132→      const max = config.max;
  1133→      if (max != null && !intInRange(max, 2, 10)) {
  1134→        return { ok: false, message: "Scale must be a whole number between 2 and 10." };
  1135→      }
  1136→      const symbol = config.symbol;
  1137→      if (
  1138→        symbol != null &&
  1139→        !["star", "heart", "thumb", "circle"].includes(String(symbol))
  1140→      ) {
  1141→        return { ok: false, message: "Unknown rating symbol." };
  1142→      }
  1143→      for (const k of ["leftLabel", "rightLabel"] as const) {
  1144→        const v = config[k];
  1145→        if (typeof v === "string" && v.length > MAX_END_LABEL_LEN) {
  1146→          return {
  1147→            ok: false,
  1148→            message: `${k === "leftLabel" ? "Left" : "Right"} label must be at most ${MAX_END_LABEL_LEN} characters.`,
  1149→          };
  1150→        }
  1151→      }
  1152→      return { ok: true };
  1153→    }
  1154→
  1155→    case "file_upload": {
  1156→      const maxSizeMb = config.maxSizeMb;
  1157→      if (
  1158→        maxSizeMb != null &&
  1159→        (!Number.isFinite(Number(maxSizeMb)) ||
  1160→          Number(maxSizeMb) <= 0 ||
  1161→          Number(maxSizeMb) > 100)
  1162→      ) {
  1163→        return { ok: false, message: "Max size must be between 0 and 100 MB." };
  1164→      }
  1165→      const allowedTypes = config.allowedTypes;
  1166→      if (
  1167→        allowedTypes != null &&
  1168→        (!Array.isArray(allowedTypes) ||
  1169→          allowedTypes.some((t) => typeof t !== "string"))
  1170→      ) {
  1171→        return { ok: false, message: "Allowed types must be a list of strings." };
  1172→      }
  1173→      return { ok: true };
  1174→    }
  1175→
  1176→    default:
  1177→      return { ok: true };
  1178→  }
  1179→}

  1181→/** Width is validated separately — the DB CHECK is 1..12. */
  1182→export function validateWidth(width: number): ConfigValidation {
  1183→  if (!Number.isInteger(width) || width < 1 || width > 12) {
  1184→    return { ok: false, message: "Width must be between 1 and 12." };
  1185→  }
  1186→  return { ok: true };
  1187→}
