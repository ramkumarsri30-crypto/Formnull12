     1→     1→/**
     2→     2→ * FormNull — Field System 2.0 Registry
     3→     3→ * =====================================================================
     4→     4→ * THE single source of truth for what a field type IS and what it can
     5→     5→ * be configured to do. Consumers:
     6→     6→ *
     7→     7→ *   - field-library.tsx       which types can be ADDED (status=active)
     8→     8→ *   - field-property-editor   which properties exist per type (schema)
     9→     9→ *   - form-renderer.tsx       renders every type, incl. legacy
    10→    10→ *   - publish/public submit   enforced by migration 006 (server side)
    11→    11→ *
    12→    12→ * ── STATUS MODEL ─────────────────────────────────────────────────────
    13→    13→ *   active : offered in the field library, full Field System 2.0
    14→    14→ *            property depth (first 10 types of the rebuild).
    15→    15→ *   legacy : NOT offered for new fields, but existing forms keep
    16→    16→ *            rendering / editing / publishing / submitting them
    17→    17→ *            through this same registry (compatibility layer — no
    18→    18→ *            destructive migration, no data loss).
    19→    19→ *
    20→    20→ * ── HONEST-PROPERTY PRINCIPLE (extends the Phase 3 rule) ────────────
    21→    21→ * Every PropertyDefinition declares where it lives and who enforces it:
    22→    22→ *
    23→    23→ *   target: "column"  → a real form_fields column (002/006 snapshot)
    24→    24→ *   target: "config"  → a key inside form_fields.config (jsonb)
    25→    25→ *
    26→    26→ *   enforcement:
    27→    27→ *     "server"       → submit_public_form (006) validates it on every
    28→    28→ *                      public submission. Fully honest validation.
    29→    29→ *     "client"       → enforced by the shared renderer only (e.g. the
    30→    30→ *                      short_text JS `pattern`, decimal `precision`).
    31→    31→ *                      Documented as a browser-side check; the server
    32→    32→ *                      still applies its own type/length backstop.
    33→    33→ *     "presentation" → changes rendering only (labels, symbols,
    34→    34→ *                      rows). Flows through the 006 snapshot to the
    35→    35→ *                      public form unchanged.
    36→    36→ *
    37→    37→ * No property is declared unless at least one of those is true. No UI
    38→    38→ * control ships without a real consumer.
    39→    39→ *
    40→    40→ * ── EXTENSION CONTRACT (Logic / Calculations / Themes / Integrations) ──
    41→    41→ * Future systems must be able to reason about fields WITHOUT touching
    42→    42→ * the field model. They consume, they do not rewrite:
    43→    43→ *   - `answerType`  : the JSON shape a submitted answer takes — Logic
    44→    44→ *                     conditions and Calculations branch on this.
    45→    45→ *   - `collectsData`: whether the field participates in submissions.
    46→    46→ *   - `publishable` : whether it can enter a published snapshot.
    47→    47→ * A "card / one-question-at-a-time" form mode consumes the exact same
    48→    48→ * field rows — presentation mode is a renderer concern, never a field
    49→    49→ * definition concern.
    50→    50→ */

   112→   112→export interface PropertyDefinition {

   911→   911→export function fieldDef(type: FieldType): FieldTypeDef {
   912→   912→  const def = FIELD_REGISTRY.find((t) => t.value === type);
   913→   913→  if (def) return def;
   914→   914→  // Unreachable while the registry covers the full DB enum; keep the
   915→   915→  // builder honest instead of crashing on future enum values.
   916→   916→  throw new Error(`Unknown field type: ${type}`);
   917→   917→}
   918→   918→
   919→   919→export function fieldDefSafe(type: FieldType): FieldTypeDef | undefined {
   920→   920→  return FIELD_REGISTRY.find((t) => t.value === type);
   921→   921→}
   922→   922→
   923→   923→/** Types offered in the field library (active only, in group order). */
   924→   924→export const ADDABLE_FIELD_TYPES: FieldTypeDef[] = FIELD_REGISTRY.filter(
   925→   925→  (t) => t.status === "active",
   926→   926→);
   927→   927→
   928→   928→export const FIELD_TYPES_BY_GROUP: Record<FieldGroup, FieldTypeDef[]> =
   929→   929→  FIELD_GROUPS.reduce(
   930→   930→    (acc, g) => {
   931→   931→      acc[g.key] = ADDABLE_FIELD_TYPES.filter((t) => t.group === g.key);
   932→   932→      return acc;
   933→   933→    },
   934→   934→    {} as Record<FieldGroup, FieldTypeDef[]>,
   935→   935→  );

   946→   946→/* ------------------------------------------------------------------ */
   947→   947→/* Config validation — mirrors 006 publish_form's checks so the       */
   948→   948→/* editor blocks invalid configs BEFORE they reach the database       */
   949→   949→/* ------------------------------------------------------------------ */
   950→   950→
   951→   951→export interface ConfigValidation {
   952→   952→  ok: boolean;
   953→   953→  message?: string;
   954→   954→}
   955→   955→
   956→   956→/**
   957→   957→ * A number 006 can round-trip: its numeric regex is
   958→   958→ * ^-?[0-9]{1,10}([.][0-9]{1,6})?$ — JS floats outside that range
   959→   959→ * (1e21, 1e-8) serialize exponentially and would be rejected at
   960→   960→ * publish time. Catch them here, honestly, up front.
   961→   961→ */
   962→   962→function isRepresentableNumber(v: unknown): boolean {
   963→   963→  if (typeof v !== "number" || !Number.isFinite(v)) return false;
   964→   964→  if (Math.abs(v) >= 1e10) return false;
   965→   965→  const s = String(v);
   966→   966→  if (s.includes("e") || s.includes("E")) return false;
   967→   967→  const decimals = s.includes(".") ? s.split(".")[1].length : 0;
   968→   968→  return decimals <= 6;
   969→   969→}
   970→   970→
   971→   971→function intInRange(v: unknown, min: number, max: number): boolean {
   972→   972→  return (
   973→   973→    typeof v === "number" && Number.isInteger(v) && v >= min && v <= max
   974→   974→  );
   975→   975→}
   976→   976→
   977→   977→/** Validate an option list (values + labels) for select-like fields. */
   978→   978→export function validateOptions(
   979→   979→  options: string[],
   980→   980→  labels: Record<string, string>,
   981→   981→): ConfigValidation {
   982→   982→  if (!Array.isArray(options) || options.length === 0) {
   983→   983→    return { ok: false, message: "Add at least one option." };
   984→   984→  }
   985→   985→  if (options.length > MAX_OPTIONS_PER_FIELD) {
   986→   986→    return {
   987→   987→      ok: false,
   988→   988→      message: `At most ${MAX_OPTIONS_PER_FIELD} options are allowed.`,
   989→   989→    };
   990→   990→  }
   991→   991→  const seen = new Set<string>();
   992→   992→  for (const v of options) {
   993→   993→    if (typeof v !== "string" || v.trim().length === 0) {
   994→   994→      return { ok: false, message: "Option values cannot be empty." };
   995→   995→    }
   996→   996→    if (v.length > MAX_OPTION_LEN) {
   997→   997→      return {
   998→   998→        ok: false,
   999→   999→        message: `Option values must be at most ${MAX_OPTION_LEN} characters.`,
  1000→  1000→      };
  1001→  1001→    }
  1002→  1002→    if (seen.has(v)) {
  1003→  1003→      return { ok: false, message: "Option values must be unique." };
  1004→  1004→    }
  1005→  1005→    seen.add(v);
  1006→  1006→  }
  1007→  1007→  for (const [value, label] of Object.entries(labels)) {
  1008→  1008→    if (typeof label !== "string") continue;
  1009→  1009→    if (label.length > MAX_PRESENTATION_LEN) {
  1010→  1010→      return {
  1011→  1011→        ok: false,
  1012→  1012→        message: `Option labels must be at most ${MAX_PRESENTATION_LEN} characters.`,
  1013→  1013→      };
  1014→  1014→    }
  1015→  1015→    if (!options.includes(value) && label.trim() !== "") {
  1016→  1016→      // Stale label entry — harmless at render (never looked up), but
  1017→  1017→      // keep configs clean.
  1018→  1018→      return { ok: false, message: "An option label has no matching value." };
  1019→  1019→    }
  1020→  1020→  }
  1021→  1021→  return { ok: true };
  1022→  1022→}
  1023→  1023→
  1024→  1024→/**
  1025→  1025→ * Validate a full config object for a type before writing it to the
  1026→  1026→ * database. Mirrors publish_form() so nothing valid here can fail
  1027→  1027→ * there, and nothing invalid here can silently pass there.
  1028→  1028→ */
  1029→  1029→export function validateConfig(
  1030→  1030→  type: FieldType,
  1031→  1031→  config: Record<string, unknown>,
  1032→  1032→): ConfigValidation {
  1033→  1033→  switch (type) {
  1034→  1034→    case "short_text":
  1035→  1035→    case "long_text":
  1036→  1036→    case "email":
  1037→  1037→    case "url":
  1038→  1038→    case "phone": {
  1039→  1039→      const minL = config.minLength;
  1040→  1040→      const maxL = config.maxLength;
  1041→  1041→      if (
  1042→  1042→        minL != null &&
  1043→  1043→        (!intInRange(minL, 0, MAX_LENGTH_CFG))
  1044→  1044→      ) {
  1045→  1045→        return {
  1046→  1046→          ok: false,
  1047→  1047→          message: `Min length must be a whole number between 0 and ${MAX_LENGTH_CFG}.`,
  1048→  1048→        };
  1049→  1049→      }
  1050→  1050→      if (maxL != null && !intInRange(maxL, 1, MAX_LENGTH_CFG)) {
  1051→  1051→        return {
  1052→  1052→          ok: false,
  1053→  1053→          message: `Max length must be a whole number between 1 and ${MAX_LENGTH_CFG}.`,
  1054→  1054→        };
  1055→  1055→      }
  1056→  1056→      if (minL != null && maxL != null && Number(minL) > Number(maxL)) {
  1057→  1057→        return { ok: false, message: "Min length cannot exceed max length." };
  1058→  1058→      }
  1059→  1059→      if (type === "short_text" && typeof config.pattern === "string" && config.pattern) {
  1060→  1060→        if (config.pattern.length > MAX_PATTERN_LEN) {
  1061→  1061→          return {
  1062→  1062→            ok: false,
  1063→  1063→            message: `Pattern must be at most ${MAX_PATTERN_LEN} characters.`,
  1064→  1064→          };
  1065→  1065→        }
  1066→  1066→        try {
  1067→  1067→          new RegExp(config.pattern);
  1068→  1068→        } catch {
  1069→  1069→          return { ok: false, message: "Pattern is not a valid regular expression." };
  1070→  1070→        }
  1071→  1071→      }
  1072→  1072→      if (type === "phone" && config.defaultCountry != null) {
  1073→  1073→        if (typeof config.defaultCountry !== "string" || !/^[A-Z]{2}$/.test(config.defaultCountry)) {
  1074→  1074→          return { ok: false, message: "Default country must be a 2-letter code." };
  1075→  1075→        }
  1076→  1076→      }
  1077→  1077→      return { ok: true };
  1078→  1078→    }
  1079→  1079→
  1080→  1080→    case "number":
  1081→  1081→    case "decimal":
  1082→  1082→    case "scale": {
  1083→  1083→      const { min, max, step } = config;
  1084→  1084→      if (min != null && !isRepresentableNumber(min)) {
  1085→  1085→        return { ok: false, message: "Min must be a number (up to 10 digits, 6 decimals)." };
  1086→  1086→      }
  1087→  1087→      if (max != null && !isRepresentableNumber(max)) {
  1088→  1088→        return { ok: false, message: "Max must be a number (up to 10 digits, 6 decimals)." };
  1089→  1089→      }
  1090→  1090→      if (step != null && (!isRepresentableNumber(step) || Number(step) <= 0)) {
  1091→  1091→        return { ok: false, message: "Step must be greater than 0 (up to 6 decimals)." };
  1092→  1092→      }
  1093→  1093→      if (min != null && max != null && Number(min) >= Number(max)) {
  1094→  1094→        return { ok: false, message: "Min must be less than max." };
  1095→  1095→      }
  1096→  1096→      if (type === "decimal" && config.precision != null) {
  1097→  1097→        if (!intInRange(config.precision, 0, 6)) {
  1098→  1098→          return { ok: false, message: "Precision must be a whole number between 0 and 6." };
  1099→  1099→        }
  1100→  1100→      }
  1101→  1101→      if (type === "scale") {
  1102→  1102→        for (const k of ["leftLabel", "rightLabel"] as const) {
  1103→  1103→          const v = config[k];
  1104→  1104→          if (typeof v === "string" && v.length > MAX_END_LABEL_LEN) {
  1105→  1105→            return {
  1106→  1106→              ok: false,
  1107→  1107→              message: `${k === "leftLabel" ? "Left" : "Right"} label must be at most ${MAX_END_LABEL_LEN} characters.`,
  1108→  1108→            };
  1109→  1109→          }
  1110→  1110→        }
  1111→  1111→      }
  1112→  1112→      return { ok: true };
  1113→  1113→    }
  1114→  1114→
  1115→  1115→    case "single_select":
  1116→  1116→    case "multi_select": {
  1117→  1117→      const options = Array.isArray(config.options)
  1118→  1118→        ? (config.options as unknown[])
  1119→  1119→        : [];
  1120→  1120→      const labels =
  1121→  1121→        config.optionLabels && typeof config.optionLabels === "object"
  1122→  1122→          ? (config.optionLabels as Record<string, string>)
  1123→  1123→          : {};
  1124→  1124→      const allStrings = options.every((o) => typeof o === "string");
  1125→  1125→      if (!allStrings) {
  1126→  1126→        return { ok: false, message: "Option values must be plain strings." };
  1127→  1127→      }
  1128→  1128→      return validateOptions(options as string[], labels);
  1129→  1129→    }
  1130→  1130→
  1131→  1131→    case "rating": {
  1132→  1132→      const max = config.max;
  1133→  1133→      if (max != null && !intInRange(max, 2, 10)) {
  1134→  1134→        return { ok: false, message: "Scale must be a whole number between 2 and 10." };
  1135→  1135→      }
  1136→  1136→      const symbol = config.symbol;
  1137→  1137→      if (
  1138→  1138→        symbol != null &&
  1139→  1139→        !["star", "heart", "thumb", "circle"].includes(String(symbol))
  1140→  1140→      ) {
  1141→  1141→        return { ok: false, message: "Unknown rating symbol." };
  1142→  1142→      }
  1143→  1143→      for (const k of ["leftLabel", "rightLabel"] as const) {
  1144→  1144→        const v = config[k];
  1145→  1145→        if (typeof v === "string" && v.length > MAX_END_LABEL_LEN) {
  1146→  1146→          return {
  1147→  1147→            ok: false,
  1148→  1148→            message: `${k === "leftLabel" ? "Left" : "Right"} label must be at most ${MAX_END_LABEL_LEN} characters.`,
  1149→  1149→          };
  1150→  1150→        }
  1151→  1151→      }
  1152→  1152→      return { ok: true };
  1153→  1153→    }
  1154→  1154→
  1155→  1155→    case "file_upload": {
  1156→  1156→      const maxSizeMb = config.maxSizeMb;
  1157→  1157→      if (
  1158→  1158→        maxSizeMb != null &&
  1159→  1159→        (!Number.isFinite(Number(maxSizeMb)) ||
  1160→  1160→          Number(maxSizeMb) <= 0 ||
  1161→  1161→          Number(maxSizeMb) > 100)
  1162→  1162→      ) {
  1163→  1163→        return { ok: false, message: "Max size must be between 0 and 100 MB." };
  1164→  1164→      }
  1165→  1165→      const allowedTypes = config.allowedTypes;
  1166→  1166→      if (
  1167→  1167→        allowedTypes != null &&
  1168→  1168→        (!Array.isArray(allowedTypes) ||
  1169→  1169→          allowedTypes.some((t) => typeof t !== "string"))
  1170→  1170→      ) {
  1171→  1171→        return { ok: false, message: "Allowed types must be a list of strings." };
  1172→  1172→      }
  1173→  1173→      return { ok: true };
  1174→  1174→    }
  1175→  1175→
  1176→  1176→    default:
  1177→  1177→      return { ok: true };
  1178→  1178→  }
  1179→  1179→}
